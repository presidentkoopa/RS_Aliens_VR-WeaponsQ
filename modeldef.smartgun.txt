Model YutaniSmartgun_3D
{
	Path "Models\Weapons\smartgun"
	Model 0 "smart_hud.md3"
	Skin 0 "smart_mode1.png"

	Scale -0.9 0.9 0.9
	Offset 0.0 -30.0 -7.0
	
	NOINTERPOLATION

	// FIXED 2026-10-23: Skin and Offset are BLOCK properties, not per-frame ones. This block
	// used to set them BETWEEN FrameIndex lines as though each applied only to the frames that
	// followed -- so the parser simply took the last of each and applied it to the whole block:
	// every frame ended up skinned "empty.png" (invisible) at Offset 0 -60 -7 (30 units out of
	// place). The Yutani display was therefore not drawn at all.
	// One skin and one offset per block means one BLOCK per skin/offset, which is what the
	// three blocks below are. Same rule the pulse rifle's counter follows with its 100 blocks.
	FrameIndex SMHD A 0 0
}
