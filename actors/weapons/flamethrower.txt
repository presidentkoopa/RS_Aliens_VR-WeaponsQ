//FLAMETHROWER

Actor M240Flamethrower_3D : M240Flamethrower replaces M240Flamethrower
{
  States
  {
  Spawn:
	WFLM A 0 A_JumpIf(GetCVAR("avw_pickupmodel")==1, "Spawn3D")
    WFLM A 1
    Loop
  Spawn3D:
	WFLM A 0 A_JumpIf(GetCVAR("avw_pickupmodel")==0, "Spawn")
    WFLM P 1 Bright
    Loop
  Ready:
    FLBF A 0 ACS_NamedExecuteAlways("LOADDOWN", 0, 20, 56)
	FLBF A 0 A_JumpifNoAmmo("LowFuelReady")
	FLBF A 0 A_PlaySound("FLAMETHR")
	FLBF A 0 A_Jumpifinventory("FlamethrowerLoaded", 40, "RealReady")
	Goto LowFuelReady
	RealReady:
	FLBF A 0 A_SetSpeed (0.7)
	FLBF A 1 A_WeaponReady(WRF_ALLOWRELOAD)
	TNT1 A 0 	
	Loop
  LowFuelReady:
    FLBF A 0 A_JumpifNoAmmo("DryFuel")
    FLBF A 1 A_WeaponReady(WRF_ALLOWRELOAD)
	FLBF A 0 	
    Loop
	Dryfuel:
	FLBF A 1 A_WeaponReady(WRF_ALLOWRELOAD)
	FLBF A 0 	
	Loop
	Dryfire:
	FLBF A 1 A_PlaySound("DRYFIRE")
	Goto Reload
  Deselect:
    FLMT A 1 A_Lower(12)
    Loop
  Select:
    FLBF A 1 A_Raise(12)
    Loop
  Fire:	
	FLMT A 0 A_JumpIfNoAmmo("Dryfire")
	FLMT A 0 A_JumpIf(waterlevel >= 3, "Dryfire")
    FLMT A 0 A_AlertMonsters
    FLMT A 0 A_Overlay(1001, "BarrelSmoke")
    FLMT A 0 bright A_GunFlash
    FLMT A 0 A_FireCustomMissile("DummyProjectile3", 0, 0, 0, 0, 0, 0)
    FLMT A 2 bright A_FireCustomMissile("FTFire", 0, 1, Random(0, 2),2)
    FLMT B 2 bright A_FireCustomMissile("FTFire", 0, 0, Random(0, 2),2)
    FLMT B 2 bright A_FireCustomMissile("FTFire", 0, 1, Random(0, 2),2)
    FLMT C 0 bright A_GunFlash
    FLMT A 0 A_FireCustomMissile("DummyProjectile3", 0, 0, 0, 0, 0, 0)
    FLMT C 2 bright A_FireCustomMissile("FTFire", 0, 1, Random(0, 2),2)
    FLMT D 2 bright A_FireCustomMissile("FTFire", 0, 0, Random(0, 2),2)
    FLMT A 2 bright A_FireCustomMissile("FTFire", 0, 1, Random(0, 2),2)
    FLMT A 0 bright A_GunFlash
    FLMT A 0 A_FireCustomMissile("DummyProjectile3", 0, 0, 0, 0, 0, 0)
    FLMT B 2 bright A_FireCustomMissile("FTFire", 0, 1, Random(0, 2),2)
    FLMT B 2 bright A_FireCustomMissile("FTFire", 0, 0, Random(0, 2),2)
    FLMT B 1 bright A_FireCustomMissile("FTFire", 0, 1, Random(0, 2),2)
    FLMT A 1 bright A_ReFire
    Goto Ready
  AltFire:
	FLMT A 0 A_PlaySound("ALIECLW1")
	FLMT A 0 A_JumpIfNoAmmo("Dryfire")
	FLMT A 0 A_FireCustomMissile("TankThrow",0,1,0,12)
	FLMT A 1 Offset(0,35) 
    FLMT A 1 Offset(0,38)
    FLMT A 1 Offset(0,44)
    FLMT A 1 Offset(0,52)
    FLMT A 1 Offset(0,52)
    FLMT A 1 Offset(0,44)
    FLMT A 1 Offset(0,38)
    FLMT A 15 Offset(0,35)
    FLMT A 0 A_ReFire
	Goto RealReady 
	 Reload:
  FLMT A 0 A_JumpIfInventory("FlamethrowerLoaded", 150, 2)
  FLMT A 0 A_JumpIfInventory("GAS", 1, "ReloadWork")
  FLMT A 0
  Goto RealReady
 ReloadWork:
 	FLMT A 0 A_GiveInventory("RSR_Go", 1)
 	Goto RSR_Start
 ReloadWork_Stock:
	/*
  FLMT A 1 Offset(0,35) A_PlaySound("MISREL")
  FLMT A 1 Offset(0,38)
  FLMT A 1 Offset(0,44)
  FLMT A 1 Offset(0,52)
  FLMT A 1 Offset(0,52)
  FLMT A 1 Offset(0,44)
  FLMT A 1 Offset(0,38)
  FLMT A 15 Offset(0,35)
  FLMT A 1 Offset(0,38) A_PlaySound("FLAMERE2")
  FLMT A 1 Offset(0,44)
  FLMT A 1 Offset(0,52)
  FLMT A 1 Offset(0,56)
  FLMT A 1 Offset(0,60)
  FLMT A 5 Offset(0,68)
  */
  FLMT E 0 A_PlaySound("MISREL")
  FLMT EFGHI 1
  FLMT J 5
 ReloadLoop:
  FLMT K 0 A_TakeInventory("GAS", 1)
  FLMT K 0 A_GiveInventory("FlamethrowerLoaded", 1)
  FLMT K 0 A_JumpIfInventory("FlamethrowerLoaded", 150, "ReloadFinish")
  FLMT K 0 A_JumpIfInventory("GAS", 1, "ReloadLoop")
  Goto ReloadFinish 
 ReloadFinish:
 /*
  TNT1 A 0 
  FLMT A 40 Offset(0,68) A_PlaySound("FLAMERE3")
  FLMT A 1 Offset(0,60)
  FLMT A 1 Offset(0,56)
  FLMT A 1 Offset(0,52)
  FLMT A 1 Offset(0,44)
  FLMT A 1 Offset(0,38)
  FLMT A 1 Offset(0,35)
  FLMT A 1 Offset(0,32)
  */
  FLMT K 24
  FLMT L 3
  FLMT MN 3
  FLMT O 3 A_PlaySound("FLAMERE2")
  FLMT PQRS 3
  FLMT T 3
  FLMT A 21 A_PlaySound("FLAMERE3")
  Goto Ready
   Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light0
    Stop
  BarrelSmoke:
	TNT1 A 0 A_JumpIf(GetCVAR("avw_weapons")==0, 7)
	TNT1 A 5
	TNT1 A 0 A_FireProjectile("FlameBarrelSmoke",0,0,0,0,FPF_NOAUTOAIM,0)
	TNT1 A 6
	TNT1 A 0 A_FireProjectile("FlameBarrelSmoke",0,0,0,0,FPF_NOAUTOAIM,0)
	TNT1 A 6
	TNT1 A 0 A_FireProjectile("FlameBarrelSmoke",0,0,0,0,FPF_NOAUTOAIM,0)
	TNT1 A 0
	Stop
  
	// RSR: real VR reload (zscript/rsr/vreload.zs, tools/rsr_patch_ae.py AE_GUNS).
	// Not in VR, or with real reload off, RSR_Start falls through to the
	// set's own canned reload and nothing below runs.
	RSR_Start:
		FLMT A 0 A_TakeInventory("RSR_Go", 1)
		FLMT A 0 A_GiveInventory("RSR_Start", 1)
		FLMT A 0 A_JumpIfInventory("RSR_Go", 1, "RSR_Out")
		Goto ReloadWork_Stock
	RSR_Out:
		FLMT A 0 A_TakeInventory("RSR_Go", 1)
		FLMT A 0 A_JumpIfInventory("RSR_NoMag_M240Flamethrower", 1, "RSR_Wait")	// no magazine in the gun
		FLMT A 0 A_GiveInventory("RSR_MagOut", 1)	// the magazine drops out (its own sound)
	RSR_Unload:	// the rounds are held aside and come back when a magazine is home, so
				// the count ends where the gun's own reload would have ended it
		FLMT A 0 A_JumpIfInventory("FlamethrowerLoaded", 1, "RSR_Unload1")
		FLMT A 0 A_GiveInventory("RSR_NoMag_M240Flamethrower", 1)
		Goto RSR_Wait
	RSR_Unload1:
		FLMT A 0 A_TakeInventory("FlamethrowerLoaded", 1)
		FLMT A 0 A_GiveInventory("RSR_Held_M240Flamethrower_3D", 1)
		Goto RSR_Unload
	RSR_Wait:		// the gun as the script shows it ("####" "#"); the script runs every tic
		"####" "#" 1 A_WeaponReady(WRF_NOBOB|WRF_NOFIRE)
		"####" "#" 0 A_GiveInventory("RSR_Tick", 1)
		"####" "#" 0 A_JumpIfInventory("RSR_Seated", 1, "RSR_Fill")
		"####" "#" 0 A_JumpIfInventory("RSR_Done", 1, "RSR_Done")
		Goto RSR_Wait
	RSR_Fill:	// the magazine is home: the gun's own fill loop, its own items
		FLMT A 0 A_TakeInventory("RSR_Seated", 1)
		FLMT A 0 A_TakeInventory("RSR_NoMag_M240Flamethrower", 1)
	RSR_Return:	// the held rounds first
		FLMT A 0 A_JumpIfInventory("RSR_Held_M240Flamethrower_3D", 1, "RSR_Return1")
		Goto RSR_FillGo
	RSR_Return1:
		FLMT A 0 A_TakeInventory("RSR_Held_M240Flamethrower_3D", 1)
		FLMT A 0 A_GiveInventory("FlamethrowerLoaded", 1)
		Goto RSR_Return
	RSR_FillGo:
		Goto RSR_FillA
	RSR_FillA:
		FLMT A 0 A_JumpIfInventory("FlamethrowerLoaded", 150, "RSR_Filled")
		FLMT A 0 A_JumpIfInventory("GAS", 1, "RSR_FillAa")
		Goto RSR_Filled
	RSR_FillAa:
		FLMT A 0 A_GiveInventory("FlamethrowerLoaded", 1)
		FLMT A 0 A_TakeInventory("GAS", 1, TIF_NOTAKEINFINITE)
		Goto RSR_FillA
	RSR_Filled:
		FLMT A 0 A_GiveInventory("RSR_Filled", 1)	// the script: the handle if due, else done
		Goto RSR_Wait
	RSR_Done:
		FLMT A 0 A_TakeInventory("RSR_Done", 1)
		Goto Ready
	}
}

		WFLM A 0 A_JumpIf(GetCVAR("avw_pickupmodel")==1, "Spawn3D")
		WFLM A 1
		Loop
	Spawn3D:
		WFLM A 0 A_JumpIf(GetCVAR("avw_pickupmodel")==0, "Spawn")
		WFLM P 1 Bright
		Loop
	Pickup.ColonistClass:
		TNT1 A 0 A_GiveInventory("M240Flamethrower_3D",1)
		Stop
	Pickup.MercenaryClass:
		TNT1 A 0 A_GiveInventory("M240Flamethrower_3D",1)
		Stop
	Pickup.YutaniClass:
		TNT1 A 0 A_GiveInventory("YutaniFlamer_3D",1)
		Stop
	Pickup.SyntheticClass:
		TNT1 A 0 A_GiveInventory("M240Flamethrower_3D",1)
		Stop
	}
